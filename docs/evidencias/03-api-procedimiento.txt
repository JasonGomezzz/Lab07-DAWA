# Evidencia del procedimiento: API tal como la deja el documento (antes de la tarea)
# Base temporal auth_db_tmp (eliminada al terminar), puerto 3099. Tokens y hashes recortados.

$ curl -s -X POST localhost:3099/api/auth/signUp -H 'Content-Type: application/json' -d '{"email":"ana@demo.pe","password":"123","name":"Ana"}'
{"id":"6ac10f4dcbc186eadbea98f8","email":"ana@demo.pe","name":"Ana"}  [HTTP 201]

# Registro público enviando roles:["admin"] en el body (el código del documento lo acepta):
$ curl -s -X POST localhost:3099/api/auth/signUp -H 'Content-Type: application/json' -d '{"email":"intruso@demo.pe","password":"123","name":"Intruso","roles":["admin"]}'
{"id":"6ac10f4dcbc186eadbea98fc","email":"intruso@demo.pe","name":"Intruso"}  [HTTP 201]

$ curl -s -X POST localhost:3099/api/auth/signUp -H 'Content-Type: application/json' -d '{"email":"ana@demo.pe","password":"otra"}'
{"message":"El email ya se encuentra en uso"}  [HTTP 400]

$ curl -s -X POST localhost:3099/api/auth/signIn -H 'Content-Type: application/json' -d '{"email":"ana@demo.pe","password":"mala"}'
{"message":"Credenciales inválidas"}  [HTTP 401]

$ curl -s -X POST localhost:3099/api/auth/signIn -H 'Content-Type: application/json' -d '{"email":"ana@demo.pe","password":"123"}'
